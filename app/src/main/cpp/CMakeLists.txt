cmake_minimum_required(VERSION 3.22.1)
project(eclipse_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp/CMakeLists.txt")
    message(FATAL_ERROR "Run scripts/setup-llama.sh first to download llama.cpp source")
endif()
add_subdirectory(vendor/llama.cpp llama_compiled)
add_library(eclipse_jni SHARED bridge.cpp)
target_include_directories(eclipse_jni PRIVATE vendor/llama.cpp/include)
target_link_libraries(eclipse_jni PRIVATE llama log android)
